Hide all sections and REMOVE names from all inputs document.querySelectorAll('.hidden-section').forEach(el => { el.style.display = 'none'; el.querySelectorAll('[data-field]').forEach(input => input.removeAttribute('name')); }); // 2. Determine target section and ADD names to its inputs let targetId = ''; if (type === 'auto') targetId = 'autoFields'; else if (type === 'home' || type === 'business') targetId = 'propertyFields'; else if (type === 'boat') targetId = 'boatFields'; else if (type === 'life' || type === 'medical' || type === 'tour') targetId = 'contactFields'; if (targetId) { document.getElementById(targetId).style.display = 'block'; document.getElementById(targetId).querySelectorAll('[data-field]').forEach(input => { input.setAttribute('name', input.getAttribute('data-field')); }); } // 3. Handle messageBox for all except contact types if (targetId && targetId !== 'contactFields') { document.getElementById('messageBox').style.display = 'block'; const noteInput = document.querySelector('[data-field="notes"]'); noteInput.setAttribute('name', 'notes'); } }
